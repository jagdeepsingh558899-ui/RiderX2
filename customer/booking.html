showToast( "Drop select karein" ); }; // CURRENT LOCATION currentLocationBtn.onclick= useCurrentLocation; // DISTANCE CALCULATION function calculateDistance( lat1, lon1, lat2, lon2 ){ const R = 6371; const dLat = (lat2-lat1) * Math.PI/180; const dLon = (lon2-lon1) * Math.PI/180; const a = Math.sin(dLat/2) * Math.sin(dLat/2) + Math.cos(lat1*Math.PI/180) * Math.cos(lat2*Math.PI/180) * Math.sin(dLon/2) * Math.sin(dLon/2); const c = 2 * Math.atan2( Math.sqrt(a), Math.sqrt(1-a) ); return R*c; } // ROUTE async function calculateRoute(){ if(!pickupCoords || !dropCoords) return; routeDistanceKm = calculateDistance( pickupCoords.lat, pickupCoords.lng, dropCoords.lat, dropCoords.lng ); distanceText.innerText = routeDistanceKm.toFixed(1) + " km"; calculateFare(); bookRideBtn.disabled=false; } // FARE function calculateFare(){ if(routeDistanceKm<=0) return; let fare=0; const hour = new Date().getHours(); if( hour>=22 || hour<6 ){ fare = routeDistanceKm * RIDERX.nightRate; } else{ if(routeDistanceKm<=10){ fare = routeDistanceKm * RIDERX.dayRateFirst10Km; } else{ fare = (10 * RIDERX.dayRateFirst10Km) + ( (routeDistanceKm-10) * RIDERX.dayRateAfter10Km ); } } fare = Math.max( 50, Math.round(fare) ); fareAmount.innerText = "₹"+fare; fareInfo.innerText = selectedService.toUpperCase() + " Ride"; } // SEARCH INPUT pickupInput.addEventListener( "input", (e)=>{ searchLocation( e.target.value, "pickup" ); }); dropInput.addEventListener( "input", (e)=>{ searchLocation( e.target.value, "drop" ); }); // BOOK RIDE bookRideBtn.onclick= async()=>{ if(!pickupCoords || !dropCoords){ showToast( "Pickup aur drop select karein", "error" ); return; } showLoading( "Searching Rider..." ); try{ const rideData={ customerId: currentUser ? currentUser.uid : "guest", pickup:{ address: pickupInput.value, lat: pickupCoords.lat, lng: pickupCoords.lng }, drop:{ address: dropInput.value, lat: dropCoords.lat, lng: dropCoords.lng }, service: selectedService, distance: routeDistanceKm, status: "searching", createdAt: serverTimestamp() }; await addDoc( collection(db,"rides"), rideData ); hideLoading(); showToast( "Ride request sent", "success" ); setTimeout(()=>{ window.location.href= "history.html"; },1500); } catch(error){ hideLoading(); console.log(error); showToast( "Ride create failed", "error" ); } }; // AUTH onAuthStateChanged( auth, (user)=>{ currentUser=user; }); // INIT window.onload=()=>{ initMap(); }; // ESCAPE HTML function escapeHtml(text){ return text .replace( //g, ">" ); } // =============================== // PART 3 START // FINAL EVENTS + RIDE BOOKING // =============================== // Reset locations function resetLocations(){ pickupCoords = null; dropCoords = null; statusText.innerText = "Move map for destination"; }); // Draw route async function drawRoute(){ if(!pickupCoords || !dropCoords) return; if(routeLayer){ map.removeLayer(routeLayer); } routeLayer = L.polyline( [ [ pickupCoords.lat, pickupCoords.lng ], [ dropCoords.lat, dropCoords.lng ] ], { weight:5 } ).addTo(map); map.fitBounds( routeLayer.getBounds(), { padding:[50,50] } ); } // Update after locations ready async function updateRide(){ await drawRoute(); await calculateRoute(); } // Book Ride final bookRideBtn.addEventListener( "click", async()=>{ if(!currentUser){ showToast( "Please login first", "error" ); hideLoading(); showToast( "Ride booked successfully", "success" ); setTimeout( ()=>{ window.location.href= "history.html"; }, 1200 ); } catch(error){ hideLoading(); console.log(error); showToast( "Booking failed", "error" ); } }); // Start app document.addEventListener( "DOMContentLoaded", ()=>{ initMap(); }
